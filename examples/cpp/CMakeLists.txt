cmake_minimum_required(VERSION 3.18)
project(crabby_native_examples LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CRABBY_LIBRARY_DIR "${CMAKE_CURRENT_LIST_DIR}/../../target/release" CACHE PATH "Directory containing the Rust cdylib")
set(CRABBY_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../include" CACHE PATH "Directory containing crabby.h and crabby.hpp")
# Use only the shared library: linking Rust staticlibs requires platform-specific dependencies.
if(WIN32)
    find_library(CRABBY_LIBRARY NAMES ponko_ffi.dll.lib ponko_ffi PATHS "${CRABBY_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
elseif(APPLE)
    find_file(CRABBY_LIBRARY NAMES libponko_ffi.dylib PATHS "${CRABBY_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
else()
    find_file(CRABBY_LIBRARY NAMES libponko_ffi.so PATHS "${CRABBY_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
endif()
add_library(ponko_ffi INTERFACE)
target_include_directories(ponko_ffi INTERFACE "${CRABBY_INCLUDE_DIR}")
target_link_libraries(ponko_ffi INTERFACE "${CRABBY_LIBRARY}")
set(CMAKE_BUILD_RPATH "${CRABBY_LIBRARY_DIR}")
function(crabby_native target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE ponko_ffi)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(WIN32)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CRABBY_LIBRARY_DIR}/ponko_ffi.dll" "$<TARGET_FILE_DIR:${target}>")
    endif()
    # Games find textures in an assets/ folder next to the executable.
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_LIST_DIR}/../../assets" "$<TARGET_FILE_DIR:${target}>/assets")
endfunction()
crabby_native(crabby_c_example main.c)
crabby_native(crabby_cpp_example main.cpp)
crabby_native(crabby_c_tests tests.c)
crabby_native(crabby_cpp_tests tests.cpp)
crabby_native(crabby_cpp_bench bench.cpp)
crabby_native(crabby_cpp_boids boids.cpp)
crabby_native(crabby_cpp_guide guide.cpp)
crabby_native(crabby_cpp_battle battle.cpp)
enable_testing()
add_test(NAME c_example COMMAND crabby_c_example)
add_test(NAME cpp_example COMMAND crabby_cpp_example)
add_test(NAME c_abi COMMAND crabby_c_tests)
add_test(NAME cpp_wrapper COMMAND crabby_cpp_tests)
add_test(NAME cpp_guide COMMAND crabby_cpp_guide)
add_test(NAME cpp_boids_headless COMMAND crabby_cpp_boids)
set_tests_properties(cpp_boids_headless PROPERTIES ENVIRONMENT "HEADLESS=1;BOIDS=2000;TICKS=30")
add_test(NAME cpp_battle_headless COMMAND crabby_cpp_battle)
set_tests_properties(cpp_battle_headless PROPERTIES ENVIRONMENT "HEADLESS=1;UNITS=2000;TICKS=600")
